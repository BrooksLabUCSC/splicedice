# Setup per run

## define location

```bash
this_commit=2b2c401
this_description=validation
this_datestamp=2026.06.23_15.07.59
this_dockerfile=Dockerfile_${this_description}_${this_commit}.txt
this_SHA=2b2c4014be8ea0e914e8dbf570b21a44ce303890
```

```bash

this_base_dir=/mnt/sd/${this_description}_${this_commit}_${this_datestamp}/
code_base=${this_base_dir}/git_code/splicedice/examples/validation_after_2b2c401/
mkdir -p ${this_base_dir}/git_code/ ${this_base_dir}/analysis/

```

(note; validation_after_2b2c401 was moved from splicedice/examples/ to splicedice/validation/processes/ after this work was performed and recorded, so some paths would need to be adjusted to reproduce this)

## get code

```bash
cd ${this_base_dir}/git_code/
git clone --depth 5 \
https://github.com/BrooksLabUCSC/splicedice.git 
cd splicedice
git reset --hard $this_SHA

```

## build docker

```bash
docker build --build-arg CACHE_BUST=$(date +%s) -t splicedice_analysis:latest -f ${code_base}/${this_dockerfile} .
bash ~/alert_msg.sh "docker build complete"

```

## identify manifests

we'll be using bed and bam files from another walkthrough since the bed and bam files are generated by code outside  the repo

```bash
bed_manifest=${this_base_dir}/git_code/splicedice/examples/SUGP1_walkthrough/bed_manifest.tsv
bam_manifest=${bed_manifest/bed/bam}

```



# Run pipeline

## run quant

```bash
date
time docker run --rm \
-v /mnt/:/mnt \
splicedice_analysis:latest \
splicedice quant -m ${bed_manifest} \
-o ${this_base_dir}/analysis/
date
 ~/alert_msg.sh "quant run complete"

```

## run intron coverage

about 25 minutes

```bash
n_threads=8
mkdir ${this_base_dir}/coverage_output/
date
time docker run --rm \
    -v /mnt/:/mnt \
    splicedice_analysis:latest \
    splicedice intron_coverage \
    -b $bam_manifest  \
    -j ${this_base_dir}/analysis/_junctions.bed \
    -n "${n_threads}" \
    -o ${this_base_dir}/coverage_output/
date
~/alert_msg.sh "intron_coverage run complete"
```

## Create intron table

about 2 minutes

```bash
genes=/mnt/ref/gencode.v47.primary_assembly.annotation.gtf
date
time docker run --rm \
-v /mnt/:/mnt \
splicedice_analysis:latest \
splicedice ir_table \
--annotation $genes \
-i ${this_base_dir}/analysis/_inclusionCounts.tsv \
-c ${this_base_dir}/analysis/_allClusters.tsv \
-d ${this_base_dir}/coverage_output \
-n 8 \
-o ${this_base_dir}/analysis/
date
```
